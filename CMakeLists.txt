cmake_minimum_required(VERSION 3.20)
project(chroma LANGUAGES CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CHROMA_BLAS "Use BLAS for matrix multiplication" ON)
option(CHROMA_METAL "Build the Apple Metal runtime" ${APPLE})

add_library(chroma_runtime INTERFACE)
target_compile_features(chroma_runtime INTERFACE cxx_std_20)
target_include_directories(chroma_runtime INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/chroma/_src")
if(CHROMA_BLAS)
    if(APPLE)
        target_compile_definitions(chroma_runtime INTERFACE CHROMA_ACCELERATE ACCELERATE_NEW_LAPACK)
        target_link_libraries(chroma_runtime INTERFACE "-framework Accelerate")
    else()
        find_package(BLAS REQUIRED)
        target_compile_definitions(chroma_runtime INTERFACE CHROMA_CBLAS)
        target_link_libraries(chroma_runtime INTERFACE BLAS::BLAS)
    endif()
endif()

if(CHROMA_METAL)
    if(NOT APPLE)
        message(FATAL_ERROR "CHROMA_METAL requires macOS")
    endif()
    enable_language(OBJCXX)
    add_library(chroma_metal INTERFACE)
    target_compile_features(chroma_metal INTERFACE cxx_std_20)
    target_include_directories(chroma_metal INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/chroma/_src")
    target_compile_options(chroma_metal INTERFACE -fobjc-arc -fblocks -mmacosx-version-min=15.0)
    target_link_options(chroma_metal INTERFACE -mmacosx-version-min=15.0)
    target_link_libraries(chroma_metal INTERFACE "-framework Metal" "-framework Foundation")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(runtime_core_test tests/runtime_core.cc)
    target_link_libraries(runtime_core_test PRIVATE chroma_runtime)
    add_test(NAME runtime_core COMMAND runtime_core_test)
    if(CHROMA_METAL)
        add_executable(runtime_metal_test tests/runtime_metal.mm)
        set_target_properties(runtime_metal_test PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
        target_link_libraries(runtime_metal_test PRIVATE chroma_metal)
        add_test(NAME runtime_metal COMMAND runtime_metal_test)
    endif()
endif()
